on_actions = {
    on_startup = {
        effect = {
            every_country = {
                limit = { is_ai = no }    # remove this limit if AI countries need these vars

                # ===== Core demographics =====
                set_variable = { total_population = max_manpower_k }

                set_variable = { workforce_share = 0.86 }       # share of population able to work
                set_variable = { g_employed = 0 }               # total employed workers
                set_variable = { g_unemployed = 0 }             # unemployed count
                set_variable = { unemployment_percent = 0 }     # unemployment rate %

                set_variable = { TAX_CORP = 0.09} # Taxes from companies/factories
                set_variable = { TAX_CIVS = 0.24} # # Taxes from civs
                set_variable = { TAX_ENERGY = 0.05} # # Taxes for energy


                set_variable = { TOTAL_MONEY = 13.7}
                set_variable = { TOTAL_DEBT = 59}

                # ===== Income & budget =====
                set_variable = { PROFIT_FROM_CFFF = 0.05}       # (Civilian) factories
                set_variable = { PROFIT_FROM_OFFF = 0.24}       # Corporate districts
                set_variable = { PROFIT_FROM_MFFF = 0.035}      # Mils
                set_variable = { PROFIT_FROM_DFFF = 0.023}      # Dockyards
                set_variable = { PROFIT_FROM_GFFF = 0.075}      # Graveyards
                set_variable = { PROFIT_FROM_SFFF = 0.12}       # Servers
                set_variable = { PROFIT_FROM_EFFF = 0.05}       # Energy
                set_variable = { PROFIT_FROM_FFFF = 0.05}       # Farm

                # ===== Expenses =====
                set_variable = { EXPENSES_ON_ARFF = 0.05}       # Army
                set_variable = { EXPENSES_ON_NAFF = 0.05}       # Navy
                set_variable = { EXPENSES_ON_AIFF = 0.05}       # Air
                set_variable = { EXPENSES_ON_EDFF = 0.05}       # Education
                set_variable = { EXPENSES_ON_PGFF = 0.05}       # Plasma generator
                set_variable = { EXPENSES_ON_CDFF = 0.05}       # Civilian districts

                # ===== Calculated / temp =====
                set_variable = { monthly_income = 0 }            # monthly income
                set_variable = { monthly_expense = 0 }           # monthly expense
                set_variable = { net_growth = 0 }               # net monthly  growth
                set_variable = { gdp_nominal = 0 }              # nominal GDP
                set_variable = { gdp_real = 0 }                 # real GDP (adjusted)
                set_variable = { average_wage = 0 }             # average wage per worker
                set_variable = { temp_calc = 0 }                # generic temp variable
            }
        }
    }
    on_monthly = {
        effect = {
            every_country = {
                # ===== Workforce =====
                set_variable = { total_population = max_manpower_k }        # total pop (thousands)
                set_variable = { g_employed = total_population }        # total pop (thousands)
                multiply_variable = { g_employed = workforce_share }  # working-age population
                set_variable = { g_unemployed = total_population }
                subtract_from_variable = { g_unemployed = g_employed } # unemployed = workforce - employed

                set_variable = { unemployment_percent = g_unemployed }
                divide_variable = { unemployment_percent = total_population }

                # ===== Income =====
                set_variable = { monthly_income = 0 }
                add_to_variable = { monthly_income = tax_income }
                add_to_variable = { monthly_income = factory_profit }
                add_to_variable = { monthly_income = office_profit }
                add_to_variable = { monthly_income = resource_export_income }



                # Resources export
                set_variable = { chromium_export = resource_exported@chromium}
			    multiply_variable = {chromium_export = 0.005}

                set_variable = { tungsten_export = resource_exported@tungsten}
			    multiply_variable = {tungsten_export = 0.005}

                set_variable = { rubber_export = resource_exported@rubber}
			    multiply_variable = {rubber_export = 0.005}

                set_variable = { aluminium_export = resource_exported@aluminium}
			    multiply_variable = {aluminium_export = 0.001}

                set_variable = { oil_export = resource_exported@oil}
			    multiply_variable = {oil_export = 0.008}

                set_variable = { steel_export = resource_exported@steel}
			    multiply_variable = {steel_export = 0.01}

                set_variable = { elysium_export = resource_exported@elysium}
			    multiply_variable = {elysium_export = 0.09}

                set_variable = { gold_export = resource_exported@gold}
			    multiply_variable = {gold_export = 0.025}

                set_variable = { plasma_export = resource_exported@plasma}
			    multiply_variable = {gold_export = 0.04}
                
			    # Resources import
                set_variable = { chromium_import = resource_imported@chromium}
			    multiply_variable = {chromium_import = 0.01}

                set_variable = { tungsten_import = resource_imported@tungsten}
			    multiply_variable = {tungsten_import = 0.01}

                set_variable = { rubber_import = resource_imported@rubber}
			    multiply_variable = {rubber_import = 0.01}

                set_variable = { aluminium_import = resource_imported@aluminium}
			    multiply_variable = {aluminium_import = 0.005}

                set_variable = { oil_import = resource_imported@oil}
			    multiply_variable = {oil_import = 0.01}

                set_variable = { steel_import = resource_imported@steel}
			    multiply_variable = {steel_import = 0.02}

                set_variable = { elysium_import = resource_imported@elysium}
			    multiply_variable = {elysium_import = 0.13}

                set_variable = { plasma_import = resource_imported@plasma}
			    multiply_variable = {elysium_import = 0.08}

                set_variable = { gold_import = resource_imported@gold}
			    multiply_variable = {gold_import = 0.05}
            }
        }
    }
}
