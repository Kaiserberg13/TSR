on_weekly = {
    effect = {
        every_country = {
            limit = { is_ai = no }

            every_owned_state = {
                limit = {
                    is_core_of = ROOT
                    OR = {
                        has_state_flag = districts_office_construction_start_construction
                        has_state_flag = districts_office_construction_in_progress
                    }
                }

                log = ">>> [ROOT.GetName] – State: [This.GetName] <<<"
                log = "Const Days: [?ROOT.OFFICE_CONSTRUCTION_DAYS] | Eff: [?ROOT.OFFICE_CONSTRUCTION_EFFICIENCY]"

                # --- ИНИЦИАЛИЗАЦИЯ ---
                if = {
                    limit = { has_state_flag = districts_office_construction_start_construction }
                    clr_state_flag = districts_office_construction_start_construction
                    set_state_flag = districts_office_construction_in_progress

                    ROOT = {
                        set_variable = { temp_total_ofd = var:OFFICE_CONSTRUCTION_DAYS }
                        multiply_variable = { temp_total_ofd = var:OFFICE_CONSTRUCTION_EFFICIENCY }
                    }

                    set_variable = { office_total_days     = var:ROOT.temp_total_ofd }
                    set_variable = { office_remaining_days = var:ROOT.temp_total_ofd }
                    set_variable = { office_progress       = 0 }

                    log = "Init total_days = [?office_total_days], remaining = [?office_remaining_days]"
                }

                # --- ПРОГРЕСС СТРОЙКИ ---
                if = {
                    limit = { has_state_flag = districts_office_construction_in_progress }

                    subtract_from_variable = { office_remaining_days = var:ROOT.OFFICE_CONSTRUCTION_DAYS }
                    if = {
                        limit = { check_variable = { office_remaining_days < 0 } }
                        set_variable = { office_remaining_days = 0 }
                    }

                    set_temp_variable = { done = var:office_total_days }
                    subtract_from_temp_variable = { done = var:office_remaining_days }

                    if = {
                        limit = { check_variable = { office_total_days > 0 } }
                        set_variable   = { office_progress = var:done }
                        divide_variable = { office_progress = var:office_total_days }
                        log = "Progress: [?office_progress] ([?done]/[?office_total_days])"
                    }

                    # --- ЗАВЕРШЕНИЕ ---
                    if = {
                        limit = { check_variable = { office_remaining_days = 0 } }
                        clr_state_flag = districts_office_construction_in_progress

                        FROM = { add_political_power = 25 }
                        log = ">>> [This.GetName] – Construction Completed <<<"

                        district_apply_office_and_sphere = yes
                    }
                }
            }
        }
    }
}
