        context_type = selected_state_context

			build_office_button_click = {
				FROM.FROM = {
					set_state_flag = districts_selected_state

					log = "Sector 2 Rotatosk[0]: [?rota_fac]"
					log = "Sector 3 Rotatosk[0]: [?rota_pro]"
				}

				if = {
					limit = {
						FROM.FROM = { 
							NOT = {
								 OR = {
									has_state_flag = districts_office_construction_sphere_military
									has_state_flag = districts_office_construction_sphere_factories
									has_state_flag = districts_office_construction_sphere_production
									has_state_flag = districts_office_construction_sphere_security

									has_state_flag = districts_office_construction_sphere_research
									has_state_flag = districts_office_construction_sphere_resources
								}
							}
						}
					}
					ROOT = {
						set_country_flag = corpo_build_office
					}
				}
			}

		}
        
        triggers = {
		}
    }
