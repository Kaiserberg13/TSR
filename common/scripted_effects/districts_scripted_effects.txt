district_apply_office_and_sphere = {

    ## Camelia (sec, fin, res, agr)
    if = { limit = { has_state_flag = districts_office_construction_camelia }
        add_to_variable = { var = camelia_office value = 1 }
        if = { limit = { has_state_flag = districts_office_construction_sphere_security }  clr_state_flag = districts_office_construction_sphere_security add_to_variable = { var = camelia_sec value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_finances }  clr_state_flag = districts_office_construction_sphere_finances add_to_variable = { var = camelia_fin value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_resources } clr_state_flag = districts_office_construction_sphere_resources add_to_variable = { var = camelia_res value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_agro }      clr_state_flag = districts_office_construction_sphere_agro add_to_variable = { var = camelia_agr value = 1 } }
        district_init_camelia_from_vars = yes
        district_init_camelia_clear_vars = yes
    }

    ## Arstotzka (sec, med, energy, research)
    if = { limit = { has_state_flag = districts_office_construction_arstotzka }
        add_to_variable = { var = arstotzka_office value = 1 }
        if = { limit = { has_state_flag = districts_office_construction_sphere_security } clr_state_flag = districts_office_construction_sphere_security add_to_variable = { var = arst_sec value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_medical }  clr_state_flag = districts_office_construction_sphere_medical add_to_variable = { var = arst_med value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_energy }   clr_state_flag = districts_office_construction_sphere_energy add_to_variable = { var = arst_en  value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_research } clr_state_flag = districts_office_construction_sphere_research add_to_variable = { var = arst_res value = 1 } }
        district_init_arstotzka_from_vars = yes
        district_init_arstotzka_clear_vars = yes
    }

    ## Rotatosk (military, factories, production)
    if = { limit = { has_state_flag = districts_office_construction_rotatosk }
        add_to_variable = { var = rotatosk_office value = 1 }
        if = { limit = { has_state_flag = districts_office_construction_sphere_military }  clr_state_flag = districts_office_construction_sphere_military add_to_variable = { var = rota_mil value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_factories } clr_state_flag = districts_office_construction_sphere_factories add_to_variable = { var = rota_fac value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_production } clr_state_flag = districts_office_construction_sphere_production add_to_variable = { var = rota_pro value = 1 } }
        district_init_rotatosk_from_vars = yes
        district_init_rotatosk_clear_vars = yes
    }

    ## Zaphkiel (production, medical, agro)
    if = { limit = { has_state_flag = districts_office_construction_zaphkiel }
        add_to_variable = { var = zaphkiel_office value = 1 }
        if = { limit = { has_state_flag = districts_office_construction_sphere_production } clr_state_flag = districts_office_construction_sphere_production add_to_variable = { var = zaph_pro value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_medical }    clr_state_flag = districts_office_construction_sphere_medical add_to_variable = { var = zaph_med value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_agro }       clr_state_flag = districts_office_construction_sphere_agro add_to_variable = { var = zaph_agr value = 1 } }
        district_init_zaphkiel_from_vars = yes
        district_init_zaphkiel_clear_vars = yes
    }

    ## Equonics (medical, energy, resources)
    if = { limit = { has_state_flag = districts_office_construction_equonics }
        add_to_variable = { var = equonics_office value = 1 }
        if = { limit = { has_state_flag = districts_office_construction_sphere_medical }   clr_state_flag = districts_office_construction_sphere_medical add_to_variable = { var = equ_med value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_energy }    clr_state_flag = districts_office_construction_sphere_energy add_to_variable = { var = equ_en  value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_resources } clr_state_flag = districts_office_construction_sphere_resources add_to_variable = { var = equ_res value = 1 } }
        district_init_equonics_from_vars = yes
        district_init_equonics_clear_vars = yes
    }

    ## Local (10 spheres; 0..9 = mil, fac, pro, sec, fin, med, agr, en, resch, resrc)
    if = { limit = { has_state_flag = districts_office_construction_local }
        add_to_variable = { var = local_office value = 1 }
        if = { limit = { has_state_flag = districts_office_construction_sphere_military }   clr_state_flag = districts_office_construction_sphere_military add_to_variable = { var = loc_0 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_factories }  clr_state_flag = districts_office_construction_sphere_factories add_to_variable = { var = loc_1 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_production } clr_state_flag = districts_office_construction_sphere_production add_to_variable = { var = loc_2 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_security }   clr_state_flag = districts_office_construction_sphere_security add_to_variable = { var = loc_3 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_finances }   clr_state_flag = districts_office_construction_sphere_finances add_to_variable = { var = loc_4 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_medical }    clr_state_flag = districts_office_construction_sphere_medical add_to_variable = { var = loc_5 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_agro }       clr_state_flag = districts_office_construction_sphere_agro add_to_variable = { var = loc_6 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_energy }     clr_state_flag = districts_office_construction_sphere_energy add_to_variable = { var = loc_7 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_research }   clr_state_flag = districts_office_construction_sphere_research add_to_variable = { var = loc_8 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_resources }  clr_state_flag = districts_office_construction_sphere_resources add_to_variable = { var = loc_9 value = 1 } }
        district_init_local_from_vars = yes
        district_init_local_clear_vars = yes
    }

    ## Foreign (10 spheres; порядок как у Local)
    if = { limit = { has_state_flag = districts_office_construction_foreign }
        add_to_variable = { var = foreign_office value = 1 }
        if = { limit = { has_state_flag = districts_office_construction_sphere_military }   clr_state_flag = districts_office_construction_sphere_military add_to_variable = { var = for_0 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_factories }  clr_state_flag = districts_office_construction_sphere_factories add_to_variable = { var = for_1 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_production } clr_state_flag = districts_office_construction_sphere_production add_to_variable = { var = for_2 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_security }   clr_state_flag = districts_office_construction_sphere_security add_to_variable = { var = for_3 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_finances }   clr_state_flag = districts_office_construction_sphere_finances add_to_variable = { var = for_4 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_medical }    clr_state_flag = districts_office_construction_sphere_medical add_to_variable = { var = for_5 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_agro }       clr_state_flag = districts_office_construction_sphere_agro add_to_variable = { var = for_6 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_energy }     clr_state_flag = districts_office_construction_sphere_energy add_to_variable = { var = for_7 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_research }   clr_state_flag = districts_office_construction_sphere_research add_to_variable = { var = for_8 value = 1 } }
        if = { limit = { has_state_flag = districts_office_construction_sphere_resources }  clr_state_flag = districts_office_construction_sphere_resources add_to_variable = { var = for_9 value = 1 } }
        district_init_foreign_from_vars = yes
        district_init_foreign_clear_vars = yes
    }
}

districts_set_corpo_scripted_effect = {
    if = {
	    limit = { FROM = {has_country_flag = corpo_build_office_rotatosk} }
	    set_state_flag = districts_office_construction_rotatosk
	}
	if = {
	    limit = { has_country_flag = corpo_build_office_zaphkiel }
	    set_state_flag = districts_office_construction_zaphkiel
	}
	if = {
	    limit = { has_country_flag = corpo_build_office_equonics }
	    set_state_flag = districts_office_construction_equonics
	}
	if = {
	    limit = { has_country_flag = corpo_build_office_arstotzka }
	    set_state_flag = districts_office_construction_arstotzka
	}
	if = {
	    limit = { has_country_flag = corpo_build_office_foreign }
	    set_state_flag = districts_office_construction_foreign
	}
	if = {
	    limit = { has_country_flag = corpo_build_office_local }
	    set_state_flag = districts_office_construction_local
	}
	if = {
	    limit = { has_country_flag = corpo_build_office_government }
	    set_state_flag = districts_office_construction_government
	}
	if = {
	    limit = { has_country_flag = corpo_build_office_camelia }
	    set_state_flag = districts_office_construction_camelia
	}
}
